# The standalone `fastmcp` package (4.x) requires mcp>=2, but langchain-mcp-adapters only
# supports mcp<2 (checked directly: 0.3.2 fails to import against mcp 2.x with
# "ImportError: cannot import name 'RequestContext'"). The mcp SDK itself ships the exact same
# FastMCP class built in (mcp.server.fastmcp.FastMCP), so server.py imports from there instead -
# same decorator API (@mcp.resource, @mcp.tool, mcp.run()), zero version conflict, one less
# dependency.
mcp<2
# catalog.py's Product TypedDict needs this directly (not just transitively via pydantic):
# plain typing.TypedDict on Python <3.12 doesn't carry the metadata Pydantic's schema
# generation needs for a Union return type (Product | dict[str, str] in server.py's
# one_product resource) - confirmed live via "Please use typing_extensions.TypedDict
# instead of typing.TypedDict on Python < 3.12" from pydantic itself.
typing-extensions>=4.12
langchain-mcp-adapters>=0.3,<0.4
langchain>=1.4,<2
